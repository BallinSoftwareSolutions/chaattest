cmake_minimum_required(VERSION 3.20)
project(GameTune LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows target required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(GameTune WIN32 src/main.cpp)
target_compile_features(GameTune PRIVATE cxx_std_17)
target_compile_definitions(GameTune PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(GameTune PRIVATE advapi32 powrprof shell32 user32 gdi32 ole32)
if(MSVC)
  target_compile_options(GameTune PRIVATE /utf-8 /W4)
endif()
if(MINGW)
  target_link_options(GameTune PRIVATE -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(GameTuneTests src/main.cpp)
  target_compile_features(GameTuneTests PRIVATE cxx_std_17)
  target_compile_definitions(GameTuneTests PRIVATE GAMETUNE_TEST UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(GameTuneTests PRIVATE advapi32 powrprof shell32 user32 gdi32 ole32)
  if(MSVC)
    target_compile_options(GameTuneTests PRIVATE /utf-8)
  endif()
  if(MINGW)
    target_link_options(GameTuneTests PRIVATE -static)
  endif()
  add_test(NAME registry_restore COMMAND GameTuneTests)
endif()

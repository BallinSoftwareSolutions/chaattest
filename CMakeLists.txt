cmake_minimum_required(VERSION 3.20)
project(Maxifying VERSION 0.5 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows target required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(MaxifyingEngine src/engine.cpp)
target_include_directories(MaxifyingEngine PUBLIC src)
target_compile_features(MaxifyingEngine PUBLIC cxx_std_17)
target_compile_definitions(MaxifyingEngine PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(MaxifyingEngine PUBLIC advapi32 powrprof ole32 user32 psapi)
add_executable(Maxifying WIN32 src/main.cpp src/ui.cpp src/bloat.cpp src/resources.rc)
target_include_directories(Maxifying PRIVATE src)
target_link_libraries(Maxifying PRIVATE MaxifyingEngine shell32 gdi32 comdlg32 dwmapi iphlpapi ws2_32 d2d1 dwrite uuid winmm)
if(MINGW)
  target_link_options(Maxifying PRIVATE -municode -static)
endif()
if(MSVC)
  target_compile_options(MaxifyingEngine PRIVATE /utf-8 /W4)
  target_compile_options(Maxifying PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(MaxifyingTests tests/engine_tests.cpp)
  target_link_libraries(MaxifyingTests PRIVATE MaxifyingEngine)
  if(MINGW)
    target_link_options(MaxifyingTests PRIVATE -static)
  endif()
  if(MSVC)
    target_compile_options(MaxifyingTests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME registry_and_profile_engine COMMAND MaxifyingTests)
  add_executable(MaxifyingMotionTests tests/motion_tests.cpp)
  target_include_directories(MaxifyingMotionTests PRIVATE src)
  target_compile_features(MaxifyingMotionTests PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(MaxifyingMotionTests PRIVATE -static)
  endif()
  add_test(NAME ui_motion COMMAND MaxifyingMotionTests)
  add_executable(MaxifyingTextTests tests/ui_text_tests.cpp)
  target_include_directories(MaxifyingTextTests PRIVATE src)
  target_compile_features(MaxifyingTextTests PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(MaxifyingTextTests PRIVATE -static)
  endif()
  if(MSVC)
    target_compile_options(MaxifyingTextTests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME ui_text COMMAND MaxifyingTextTests)
  add_executable(MaxifyingGamingPolicyTests tests/gaming_policy_tests.cpp)
  target_include_directories(MaxifyingGamingPolicyTests PRIVATE src)
  target_compile_features(MaxifyingGamingPolicyTests PRIVATE cxx_std_17)
  if(MINGW)
    target_link_options(MaxifyingGamingPolicyTests PRIVATE -static)
  endif()
  if(MSVC)
    target_compile_options(MaxifyingGamingPolicyTests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME gaming_policy COMMAND MaxifyingGamingPolicyTests)
endif()

cmake_minimum_required(VERSION 3.20)
project(GameTune VERSION 0.2 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows target required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(GameTuneEngine src/engine.cpp)
target_include_directories(GameTuneEngine PUBLIC src)
target_compile_features(GameTuneEngine PUBLIC cxx_std_17)
target_compile_definitions(GameTuneEngine PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(GameTuneEngine PUBLIC advapi32 powrprof ole32 user32)
add_executable(GameTune WIN32 src/main.cpp src/resources.rc)
target_include_directories(GameTune PRIVATE src)
target_link_libraries(GameTune PRIVATE GameTuneEngine shell32 gdi32 comdlg32 dwmapi iphlpapi ws2_32)
if(MINGW)
  target_link_options(GameTune PRIVATE -municode -static)
endif()
if(MSVC)
  target_compile_options(GameTuneEngine PRIVATE /utf-8 /W4)
  target_compile_options(GameTune PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(GameTuneTests tests/engine_tests.cpp)
  target_link_libraries(GameTuneTests PRIVATE GameTuneEngine)
  if(MINGW)
    target_link_options(GameTuneTests PRIVATE -static)
  endif()
  if(MSVC)
    target_compile_options(GameTuneTests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME registry_and_profile_engine COMMAND GameTuneTests)
endif()
